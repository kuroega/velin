cmake_minimum_required(VERSION 3.24)
project(Velin VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(CTest)
option(VELIN_BUILD_PLAYER "Build the native Windows player" ON)
set(VELIN_MPV_DLL "${CMAKE_CURRENT_SOURCE_DIR}/runtime/libmpv-2.dll" CACHE FILEPATH
    "Path to a compatible x64 libmpv-2.dll (client API major version 2)")

add_library(velin_options INTERFACE)
target_compile_features(velin_options INTERFACE cxx_std_20)
target_compile_definitions(velin_options INTERFACE VELIN_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(velin_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
else()
    target_compile_options(velin_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_compile_definitions(velin_options INTERFACE
        UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
endif()

add_library(velin_core STATIC src/core.cpp src/player_options.cpp src/library.cpp)
target_include_directories(velin_core PUBLIC include PRIVATE third_party)
target_link_libraries(velin_core PUBLIC velin_options)

function(velin_portable_runtime target)
    if(MINGW)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endfunction()

function(velin_copy_mpv target)
    if(EXISTS "${VELIN_MPV_DLL}")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${VELIN_MPV_DLL}" "$<TARGET_FILE_DIR:${target}>/libmpv-2.dll"
            VERBATIM)
    endif()
endfunction()

if(VELIN_BUILD_PLAYER)
    if(NOT WIN32)
        message(FATAL_ERROR "The player requires Windows. Use -DVELIN_BUILD_PLAYER=OFF for core tests on other platforms.")
    endif()
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Velin currently supports 64-bit Windows builds only.")
    endif()
    enable_language(RC)
    add_library(velin_engine STATIC src/media_engine.cpp)
    target_include_directories(velin_engine PUBLIC include PRIVATE third_party)
    target_link_libraries(velin_engine PUBLIC velin_options)

    include(cmake/Extensions.cmake)
    add_library(velin_platform STATIC src/platform.cpp src/platform_ocr.cpp src/accessibility.cpp src/thumbnails.cpp)
    if(MINGW)
        # GCC 16's devirtualization miscompiles calls through OS-owned WinRT
        # vtables in the OCR bridge. Keep normal optimization for other code.
        # The real OCR regression test exercises this Release-build path.
        set_source_files_properties(src/platform_ocr.cpp src/plugin_webview.cpp PROPERTIES COMPILE_OPTIONS "-fno-devirtualize")
    endif()
    target_include_directories(velin_platform PUBLIC include)
    target_link_libraries(velin_platform PUBLIC velin_core velin_options
        PRIVATE runtimeobject uiautomationcore oleacc ole32 oleaut32 uuid propsys shell32 shlwapi
        user32 gdi32 advapi32 bcrypt windowscodecs)

    add_executable(velin WIN32 src/main.cpp src/player_window.cpp resources/velin.rc)
    foreach(_velin_window_source IN ITEMS src/plugin_webview.cpp src/tool_windows.cpp src/preferences_dialog.cpp src/player_features.cpp src/player_actions.cpp src/player_services.cpp src/player_alist.cpp)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${_velin_window_source}")
            target_sources(velin PRIVATE "${_velin_window_source}")
        endif()
    endforeach()
    target_include_directories(velin PRIVATE resources third_party)
    target_link_libraries(velin PRIVATE velin_core velin_engine velin_platform velin_extensions
        d2d1 dwrite dwmapi comctl32 comdlg32 ole32 shell32 shlwapi uuid windowscodecs user32 gdi32)
    if(MINGW)
        target_link_options(velin PRIVATE -municode)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            # WinLibs GCC 16 ships a non-default default-manifest.o, which
            # conflicts with application manifests (winlibs_mingw issue #299).
            # Override only this target's endfile spec; never modify the toolchain.
            execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpspecs
                OUTPUT_VARIABLE _velin_gcc_specs RESULT_VARIABLE _velin_specs_result)
            if(_velin_specs_result EQUAL 0)
                string(REGEX MATCH "\\*endfile:[\r\n]+([^\r\n]+)" _velin_endfile_match "${_velin_gcc_specs}")
                set(_velin_endfile "${CMAKE_MATCH_1}")
                set(_velin_default_manifest "%{!shared:%:if-exists(default-manifest.o%s)}")
                string(FIND "${_velin_endfile}" "${_velin_default_manifest}" _velin_manifest_offset)
                if(NOT _velin_manifest_offset EQUAL -1)
                    string(REPLACE "${_velin_default_manifest}" "" _velin_endfile "${_velin_endfile}")
                    set(_velin_specs_file "${CMAKE_CURRENT_BINARY_DIR}/velin-manifest.specs")
                    file(WRITE "${_velin_specs_file}" "*endfile:\n${_velin_endfile}\n\n")
                    target_link_options(velin PRIVATE "-specs=${_velin_specs_file}")
                endif()
            endif()
        endif()
    elseif(MSVC)
        # The application manifest is explicitly embedded by velin.rc.
        target_link_options(velin PRIVATE /MANIFEST:NO)
    endif()
    velin_portable_runtime(velin)
    velin_copy_mpv(velin)
    foreach(_velin_helper IN ITEMS WebView2Loader.dll ffmpeg.exe yt-dlp.exe)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/${_velin_helper}")
            add_custom_command(TARGET velin POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${CMAKE_CURRENT_SOURCE_DIR}/runtime/${_velin_helper}" "$<TARGET_FILE_DIR:velin>/${_velin_helper}" VERBATIM)
            install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/runtime/${_velin_helper}" DESTINATION .)
        endif()
    endforeach()
    add_executable(velin-cli src/cli.cpp)
    target_link_libraries(velin-cli PRIVATE velin_platform)
    velin_portable_runtime(velin-cli)
    if(NOT EXISTS "${VELIN_MPV_DLL}")
        message(STATUS "libmpv runtime is not installed. Run powershell -File scripts/Get-Mpv.ps1 before playback.")
    endif()
    install(TARGETS velin velin-cli RUNTIME DESTINATION .)
    if(EXISTS "${VELIN_MPV_DLL}")
        install(FILES "${VELIN_MPV_DLL}" DESTINATION . RENAME libmpv-2.dll)
    endif()
    install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
    install(FILES config/mpv-runtime.json DESTINATION config)
    install(DIRECTORY docs DESTINATION .)
    install(DIRECTORY browser DESTINATION .)
    install(FILES scripts/Apply-Update.ps1 DESTINATION scripts)
endif()

if(BUILD_TESTING)
    if(VELIN_BUILD_PLAYER)
        add_executable(velin_platform_smoke tests/platform_smoke.cpp)
        target_link_libraries(velin_platform_smoke PRIVATE velin_platform oleacc oleaut32 user32 gdi32)
        velin_portable_runtime(velin_platform_smoke)
        add_test(NAME platform COMMAND velin_platform_smoke)
        add_executable(velin_thumbnail_tests tests/thumbnail_tests.cpp)
        target_link_libraries(velin_thumbnail_tests PRIVATE velin_platform)
        velin_portable_runtime(velin_thumbnail_tests)
        add_test(NAME thumbnails COMMAND velin_thumbnail_tests "${CMAKE_CURRENT_SOURCE_DIR}/runtime/ffmpeg.exe")
        set_tests_properties(thumbnails PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
    endif()
    add_executable(velin_core_tests tests/core_tests.cpp)
    target_link_libraries(velin_core_tests PRIVATE velin_core)
    velin_portable_runtime(velin_core_tests)
    add_test(NAME core COMMAND velin_core_tests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/library_tests.cpp")
        add_executable(velin_library_tests tests/library_tests.cpp)
        target_link_libraries(velin_library_tests PRIVATE velin_core)
        velin_portable_runtime(velin_library_tests)
        add_test(NAME library COMMAND velin_library_tests)
    endif()
    if(VELIN_BUILD_PLAYER AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_smoke.cpp")
        add_executable(velin_engine_smoke tests/engine_smoke.cpp)
        target_link_libraries(velin_engine_smoke PRIVATE velin_engine velin_core user32 shell32 ole32)
        velin_portable_runtime(velin_engine_smoke)
        velin_copy_mpv(velin_engine_smoke)
        add_test(NAME engine_smoke COMMAND velin_engine_smoke)
        set_tests_properties(engine_smoke PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()

set(CPACK_SOURCE_IGNORE_FILES "/build/;/out/;/runtime/;/_CPack_Packages/;/\\.cache/;/\\.git/;\\.user$;\\.zip$")
